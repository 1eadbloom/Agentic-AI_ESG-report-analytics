# Install Ragas if not already installed
# pip install ragas datasets

from ragas.testset import TestsetGenerator
from datasets import Dataset

# Step 1: Prepare your source documents
# Each document should have a 'text' field
docs = [
    {"text": "Python is a popular programming language created by Guido van Rossum."},
    {"text": "The capital of France is Paris, known for the Eiffel Tower."},
    {"text": "The Great Wall of China is one of the longest structures ever built."}
]

# Convert to HuggingFace Dataset
dataset = Dataset.from_list(docs)

# Step 2: Initialize the TestsetGenerator
# You can configure parameters like number of queries per document
generator = TestsetGenerator(
    num_queries_per_doc=2,  # Generate 2 queries per document
    max_query_length=128    # Limit query length
)

# Step 3: Generate the synthetic test set
try:
    testset = generator.generate(dataset)
except Exception as e:
    print(f"Error generating testset: {e}")
    exit(1)

# Step 4: Inspect the generated test set
print("Generated Testset:")
for sample in testset:
    print(f"Query: {sample['query']}")
    print(f"Answer: {sample['answer']}")
    print(f"Context: {sample['context']}")
    print("-" * 40)

# Step 5: Save the test set for later use
testset.save_to_disk("./ragas_testset")
